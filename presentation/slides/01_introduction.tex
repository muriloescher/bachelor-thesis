% Intro / motivation

\begin{frame}
    \frametitle{Motivation}
    \begin{PraesentationAufzaehlung}
        \item Many languages are \textbf{low-resource} but morphologically rich
        \item Morphology matters for NLP: tagging, parsing, MT, generation
        \item Challenge: \textbf{data scarcity} + \textbf{large label spaces} (MSD)
    \end{PraesentationAufzaehlung}
\end{frame}

\begin{frame}
    \frametitle{Tasks}
    \begin{PraesentationAufzaehlung}
        \item \textbf{Inflection}: $(\text{lemma},\ \text{MSD}) \rightarrow \text{surface form}$
        \item \textbf{Analysis}: $\text{surface form} \rightarrow (\text{lemma},\ \text{MSD})$
        \item We study both directions across multiple languages
    \end{PraesentationAufzaehlung}
\end{frame}

\begin{frame}
    \frametitle{Contributions (high level)}
    \begin{PraesentationAufzaehlung}
        \item A unified ByT5-based approach for inflection and analysis
        \item Bidirectional training setups and comparisons
        \item Evaluation on UniMorph and UD-derived data
    \end{PraesentationAufzaehlung}
\end{frame}
