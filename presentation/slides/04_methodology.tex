\begin{frame}
    \frametitle{Directionality setups}
    \begin{PraesentationAufzaehlung}

        \item \textbf{Bidirectional}: mix both directions in training
        \item Goal: shared representations across tasks
    \end{PraesentationAufzaehlung}
\end{frame}
